CREATE DATABASE SalesAnalyticsDB;

Use SalesAnalyticsDB;


create table sales_transaction(
	transaction_id INT PRIMARY KEY,
	customer_name VARCHAR(50) NOT NULL,
	product_name VARCHAR(50) NOT NULL,
	category VARCHAR(50) NOT NULL,qauntity INT,
	unit_price INT,
	discount_percent INT,
	city VARCHAR(50) NOT NULL,
	payment_mode VARCHAR(50) NOT NULL,
	salesperson VARCHAR(50) NOT NULL,
	customer_type VARCHAR(50) NOT NULL
);

select * 
from sales_transaction
order by unit_price , qauntity DESC;

SELECT customer_name,product_name,category as Category,qauntity as Quantity,unit_price as Unit_Price,city as City 
FROM sales_transaction
WHERE unit_price > 50000 AND qauntity > 1;


select * 
from sales_transaction
where customer_type='Premium' and unit_price>25000
order by unit_price DESC;

SELECT *
FROM sales_transaction
WHERE discount_percent >15 AND qauntity >= 3
order by discount_percent DESC;

SELECT customer_name,product_name,category as Category, qauntity as Quantity,unit_price as Unit_Price,city as City
FROM sales_transaction
WHERE city IN ('Ahmedabad','Mumbai','Delhi') AND unit_price > 20000
order by discount_percent DESC;

SELECT customer_name,product_name,category as Category, qauntity as Quantity,unit_price as Unit_Price,payment_mode as Payment_Mode
FROM sales_transaction
WHERE payment_mode IN ('Online','Card') AND qauntity > 2 and unit_price >15000
order by qauntity DESC;

SELECT *
FROM sales_transaction
WHERE category IN ('Electronics','Furniture','Appliances') AND discount_percent < 10
order by discount_percent ;

SELECT customer_name,customer_type,product_name,qauntity as Quantity,unit_price as Unit_Price
FROM sales_transaction
WHERE customer_type IN ('Regular','Premium') AND unit_price>30000 AND qauntity>1
order by unit_price DESC ;

select * 
from sales_transaction
where qauntity>4 AND discount_percent<20

SELECT transaction_id,customer_name,product_name,qauntity as Quantity,unit_price as Unit_Price,category as Category
FROM sales_transaction
where (qauntity>5 and unit_price>10000) OR 
(qauntity>=2 and qauntity<=5 and unit_price>50000)
order by unit_price DESC ;
